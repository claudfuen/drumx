cmake_minimum_required(VERSION 3.20)
project(DrumxCore VERSION 0.1.0 LANGUAGES C CXX)

include(CTest)
option(DRUMX_ENABLE_SANITIZERS "Run portable checks with address and undefined-behavior sanitizers" OFF)

add_library(drumx_core STATIC native/core/drumx_core.cpp native/core/drumx_tempo.cpp)
add_library(Drumx::Core ALIAS drumx_core)
target_include_directories(drumx_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/native/core")
target_compile_features(drumx_core PUBLIC cxx_std_17)
set_target_properties(drumx_core PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)

function(drumx_check_options target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive-)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
  if(DRUMX_ENABLE_SANITIZERS)
    if(MSVC)
      message(FATAL_ERROR "DRUMX_ENABLE_SANITIZERS currently supports Clang/GCC only")
    endif()
    target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
endfunction()
drumx_check_options(drumx_core)

if(BUILD_TESTING)
  add_executable(drumx_core_checks native/core/tests.cpp)
  target_link_libraries(drumx_core_checks PRIVATE Drumx::Core)
  drumx_check_options(drumx_core_checks)
  add_test(NAME core_contract COMMAND drumx_core_checks)

  add_executable(drumx_tempo_checks native/core/tempo_tests.cpp)
  target_link_libraries(drumx_tempo_checks PRIVATE Drumx::Core)
  drumx_check_options(drumx_tempo_checks)
  add_test(NAME tempo_contract COMMAND drumx_tempo_checks)

  # Compile the consumer as C, then link it to the C++ implementation.
  add_executable(drumx_c_api_check spikes/portable-core/c_api_check.c)
  target_link_libraries(drumx_c_api_check PRIVATE Drumx::Core)
  drumx_check_options(drumx_c_api_check)
  add_test(NAME c_api_contract COMMAND drumx_c_api_check)
  set_tests_properties(core_contract tempo_contract c_api_contract PROPERTIES TIMEOUT 30)
endif()
