cmake_minimum_required(VERSION 3.20)
if(POLICY CMP0135)
  cmake_policy(SET CMP0135 NEW)
endif()
project(DrumxGameNative LANGUAGES C CXX)
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
endif()
include(FetchContent)
include(CTest)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded" CACHE STRING "Static Windows runtime" FORCE)
set(GODOTCPP_USE_STATIC_CPP ON CACHE BOOL "" FORCE)
set(GODOTCPP_TARGET template_release CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
FetchContent_Declare(godot_cpp GIT_REPOSITORY https://github.com/godotengine/godot-cpp.git
  GIT_TAG e83fd0904c13356ed1d4c3d09f8bb9132bdc6b77 GIT_SHALLOW FALSE)
FetchContent_Declare(miniaudio URL https://github.com/mackron/miniaudio/archive/f40cf03f80cdb7e741d43e53b7e706e8c1394bcf.tar.gz SOURCE_SUBDIR drumx-header-only)
FetchContent_MakeAvailable(godot_cpp miniaudio)
set(DRUMX_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../..")
add_library(drumx_backend STATIC backend.cpp native_audio.cpp native_midi.cpp "${DRUMX_ROOT}/native/core/drumx_core.cpp" "${DRUMX_ROOT}/native/core/drumx_tempo.cpp")
target_include_directories(drumx_backend PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" "${DRUMX_ROOT}/native/core" PRIVATE "${miniaudio_SOURCE_DIR}")
if(APPLE)
  target_link_libraries(drumx_backend PUBLIC "-framework CoreMIDI" "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox")
elseif(WIN32)
  target_compile_definitions(drumx_backend PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(drumx_backend PUBLIC winmm ole32)
else()
  target_link_libraries(drumx_backend PUBLIC dl pthread m)
endif()
add_library(drumx_engine SHARED drumx_engine.cpp register_types.cpp)
add_library(drumx_progress_lock STATIC progress_lock.cpp)
target_include_directories(drumx_progress_lock PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(drumx_engine PRIVATE godot::cpp drumx_backend drumx_progress_lock)
if(APPLE)
  target_sources(drumx_engine PRIVATE window_appearance_macos.mm)
  target_link_libraries(drumx_engine PRIVATE "-framework AppKit")
endif()
set_target_properties(drumx_engine PROPERTIES PREFIX "" OUTPUT_NAME "drumx_engine"
  LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../bin"
  RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/../bin")
foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(drumx_engine PROPERTIES
    LIBRARY_OUTPUT_DIRECTORY_${configuration} "${CMAKE_CURRENT_SOURCE_DIR}/../bin"
    RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_CURRENT_SOURCE_DIR}/../bin")
endforeach()
if(BUILD_TESTING)
  add_executable(drumx_progress_lock_checks progress_lock_checks.cpp)
  target_link_libraries(drumx_progress_lock_checks PRIVATE drumx_progress_lock)
  add_test(NAME shared_progress_ownership COMMAND drumx_progress_lock_checks)
  add_executable(drumx_backend_checks backend_checks.cpp)
  target_link_libraries(drumx_backend_checks PRIVATE drumx_backend)
  add_test(NAME shared_native_backend COMMAND drumx_backend_checks "${DRUMX_ROOT}/native/assets/BigRusty")
  if(APPLE)
    add_executable(drumx_coremidi_checks coremidi_checks.cpp)
    target_link_libraries(drumx_coremidi_checks PRIVATE drumx_backend)
    add_test(NAME native_coremidi_loopback COMMAND drumx_coremidi_checks)
    set_tests_properties(native_coremidi_loopback PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 10)
  endif()
endif()
